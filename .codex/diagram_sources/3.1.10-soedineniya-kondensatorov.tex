\newcommand{\capacitor}[4]{
  \draw[draw=#4,line width=0.75mm] (#1,#2-13) -- (#1,#2+13);
  \draw[draw=#4,line width=0.75mm] (#1+8,#2-13) -- (#1+8,#2+13);
  \node[font=\sffamily\bfseries\fontsize{12}{15}\selectfont,text=#4] at (#1+4,#2+23) {$#3$};
}
\newcommand{\parallelcapacitor}[4]{
  \draw[draw=#4,line width=0.75mm] (#1,#2-13) -- (#1,#2+13);
  \draw[draw=#4,line width=0.75mm] (#1+8,#2-13) -- (#1+8,#2+13);
  \node[font=\sffamily\bfseries\fontsize{10}{12}\selectfont,text=#4,fill=#4!8,inner sep=0.35mm]
    at (#1+4,#2) {$#3$};
}
