\newcommand{\outofpage}[3]{%
  \draw[draw=#3,line width=0.7mm,fill=white] (#1,#2) circle[radius=2.6mm];
  \fill[#3] (#1,#2) circle[radius=0.85mm];
}
\newcommand{\intopage}[3]{%
  \draw[draw=#3,line width=0.7mm,fill=white] (#1,#2) circle[radius=2.6mm];
  \draw[draw=#3,line width=0.7mm] ($(#1,#2)+(-1.6,-1.6)$) -- ($(#1,#2)+(1.6,1.6)$);
  \draw[draw=#3,line width=0.7mm] ($(#1,#2)+(-1.6,1.6)$) -- ($(#1,#2)+(1.6,-1.6)$);
}
