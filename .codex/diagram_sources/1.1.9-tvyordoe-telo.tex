\tikzset{
  card box/.style={fill=card,draw=border,line width=1.6pt,rounded corners=18pt},
  rigid body/.style={draw=ink!82,line width=2.7pt,fill=translation!9,rounded corners=7pt},
  ghost body/.style={draw=ink!55,line width=2.5pt,dashed,fill=translation!4,rounded corners=7pt},
  path A/.style={draw=pointA!75,line width=2.7pt,dashed,-{Stealth[length=11pt,width=8pt]}},
  path B/.style={draw=pointB!75,line width=2.7pt,dashed,-{Stealth[length=11pt,width=8pt]}},
  rotation path/.style={draw=rotation,line width=3.2pt,-{Stealth[length=12pt,width=9pt]}},
  orbit A/.style={draw=pointA!55,line width=2.3pt,dash pattern=on 9pt off 7pt},
  orbit B/.style={draw=pointB!55,line width=2.3pt,dash pattern=on 9pt off 7pt}
}
